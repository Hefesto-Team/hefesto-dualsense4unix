Os trechos exatos das abas 01 e 02. Trocar NN pelo número da mudança na aba (o n do data-todo), e ESQ/TOPO pelo vazio da aba onde o menu aparece aberto.

2. Menu do botão direito (O-BOTAO-DIREITO-ABRE-O-MENU-DO-HEFESTO-EM-TODA-ABA-01), no fim do body

<style data-todo-css="menu-do-botao-direito">
  .todo-menu{position:fixed;z-index:9999;display:none;min-width:168px;padding:4px;
    border:1px solid var(--border-forte);border-radius:8px;background:var(--elevated);
    box-shadow:0 6px 18px rgba(0,0,0,.45);font-family:'Space Grotesk',system-ui,sans-serif;font-size:13px}
  .todo-menu.aberto{display:block}
  .todo-menu button{display:block;width:100%;height:28px;padding:0 14px;border:0;border-radius:5px;
    background:transparent;color:var(--fg);font:inherit;text-align:left;cursor:pointer}
  .todo-menu button:hover{background:var(--sel-bg)}
  .todo-menu button[aria-disabled="true"]{color:var(--texto-mudo);cursor:default;background:transparent}
  .todo-menu hr{border:0;border-top:1px solid var(--border-sutil);margin:4px 2px}
</style>
<div class="todo-menu aberto" id="todo-menu-direito" data-todo="NN" role="menu" aria-label="Menu do Hefesto"
     style="left:ESQpx;top:TOPOpx"
     title="O menu de verdade é o do sistema, no tema do COSMIC; aqui ele aparece aberto só para se ver.">
  <button role="menuitem" data-item="copiar" aria-disabled="true">Copiar</button>
  <hr>
  <button role="menuitem" data-item="atualizar">Atualizar</button>
  <button role="menuitem" data-item="ligar-desligar">Desligar</button>
  <button role="menuitem" data-item="reiniciar">Reiniciar</button>
  <hr>
  <button role="menuitem" data-item="fechar">Fechar</button>
</div>
<script>
(() => {
  const m = document.getElementById('todo-menu-direito');
  const rotulo = () => {
    const desligado = document.body.classList.contains('nativo') || !!document.querySelector('#hef-desligado:checked');
    m.querySelector('[data-item="ligar-desligar"]').textContent = desligado ? 'Ligar' : 'Desligar';
  };
  document.addEventListener('contextmenu', (e) => {
    e.preventDefault();
    m.querySelector('[data-item="copiar"]').setAttribute('aria-disabled', String(!String(getSelection()).trim()));
    rotulo();
    m.classList.add('aberto');
    m.style.left = Math.min(e.clientX, innerWidth - m.offsetWidth - 8) + 'px';
    m.style.top = Math.min(e.clientY, innerHeight - m.offsetHeight - 8) + 'px';
  });
  document.addEventListener('click', (e) => { if (!m.contains(e.target)) m.classList.remove('aberto'); rotulo(); });
  document.addEventListener('change', rotulo);
  rotulo();
  document.addEventListener('keydown', (e) => { if (e.key === 'Escape') m.classList.remove('aberto'); });
})();
</script>
